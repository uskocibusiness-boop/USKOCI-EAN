DO $do$
DECLARE
  v_cat_id uuid;
  v_parent_id uuid := 'df30eb84-47fb-41ae-90f3-3f980d8abd45'; -- Digitalne usluge
  v_worker_acc uuid := '0640a1bd-5e0a-4462-9dca-f49420582c8a';
BEGIN
  -- 1. Ensure category in private.work_taxonomy_categories
  SELECT id INTO v_cat_id FROM private.work_taxonomy_categories WHERE canonical_key = '3d printovanje';
  IF v_cat_id IS NULL THEN
    INSERT INTO private.work_taxonomy_categories(id, canonical_key, label, parent_id, state, source, catalog_version)
    VALUES (gen_random_uuid(), '3d printovanje', '3D printovanje', v_parent_id, 'ESTABLISHED', 'SYSTEM', 1)
    RETURNING id INTO v_cat_id;
  END IF;

  -- 2. Ensure execution modes (both ON_SITE and REMOTE)
  INSERT INTO private.work_taxonomy_category_modes(category_id, execution_scope, source_ref)
  VALUES (v_cat_id, 'ON_SITE', 'DEV_QA_MANUAL'), (v_cat_id, 'REMOTE', 'DEV_QA_MANUAL')
  ON CONFLICT DO NOTHING;

  -- 3. Ensure aliases with locale 'sr'
  INSERT INTO private.work_taxonomy_aliases(category_id, label, locale)
  VALUES 
    (v_cat_id, '3D štampa', 'sr'),
    (v_cat_id, 'Izrada 3D figurica', 'sr'),
    (v_cat_id, '3D modelovanje i štampa', 'sr')
  ON CONFLICT DO NOTHING;

  -- 4. Bind worker capability
  INSERT INTO private.work_taxonomy_worker_capabilities(
    worker_account_id, category_id, confirmed_at, allow_related_offers, excluded, origin
  ) VALUES (
    v_worker_acc, v_cat_id, statement_timestamp(), true, false, 'WORKER_CONFIRMED'
  )
  ON CONFLICT (worker_account_id, category_id) DO UPDATE
    SET excluded = false,
        allow_related_offers = true,
        updated_at = statement_timestamp();

END $do$;
